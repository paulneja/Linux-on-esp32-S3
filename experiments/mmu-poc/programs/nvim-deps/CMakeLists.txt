cmake_minimum_required(VERSION 3.16)
project(esp32_nvim_dependencies C)
add_compile_definitions(_GNU_SOURCE)
set(SOURCES "${CMAKE_CURRENT_LIST_DIR}/../../out/programs/nvim-sources")
set(UCLIBC "${CMAKE_CURRENT_LIST_DIR}/../../out/programs/uclibc-ng-1.0.48")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(LIBUV_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LIBUV_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/compat")
configure_file("${UCLIBC}/include/ifaddrs.h" "${CMAKE_BINARY_DIR}/compat/ifaddrs.h" COPYONLY)
add_subdirectory("${SOURCES}/LIBUV" libuv)
target_sources(uv_a PRIVATE "${UCLIBC}/libc/inet/ifaddrs.c")
target_include_directories(uv_a PRIVATE "${CMAKE_BINARY_DIR}/compat")
target_compile_options(uv_a PRIVATE -include "${CMAKE_CURRENT_LIST_DIR}/uclibc-compat.h")
set_source_files_properties("${UCLIBC}/libc/inet/ifaddrs.c" PROPERTIES
  COMPILE_OPTIONS "-include;${CMAKE_CURRENT_LIST_DIR}/uclibc-compat.h")
target_link_libraries(uv_a -l:libfork.so.0)
add_library(util STATIC "${UCLIBC}/libutil/forkpty.c" "${UCLIBC}/libutil/openpty.c" "${UCLIBC}/libutil/login_tty.c")
target_compile_options(util PRIVATE -include "${CMAKE_CURRENT_LIST_DIR}/uclibc-compat.h")
install(TARGETS util ARCHIVE DESTINATION lib)

file(GLOB LUA_FILES "${SOURCES}/LUA/src/*.c")
list(FILTER LUA_FILES EXCLUDE REGEX "/(lua|luac|print)\\.c$")
add_library(lua STATIC ${LUA_FILES})
target_compile_definitions(lua PRIVATE LUA_USE_POSIX LUA_USE_DLOPEN)
target_include_directories(lua PUBLIC "${SOURCES}/LUA/src")
target_link_libraries(lua PUBLIC m dl)
install(TARGETS lua ARCHIVE DESTINATION lib)
install(FILES "${SOURCES}/LUA/src/lua.h" "${SOURCES}/LUA/src/lualib.h" "${SOURCES}/LUA/src/lauxlib.h" "${SOURCES}/LUA/src/luaconf.h" DESTINATION include)

file(GLOB LPEG_FILES "${SOURCES}/LPEG/*.c")
add_library(lpeg STATIC ${LPEG_FILES})
target_include_directories(lpeg PRIVATE "${SOURCES}/LUA/src")
install(TARGETS lpeg ARCHIVE DESTINATION lib)
add_subdirectory("${SOURCES}/UNIBILIUM" unibilium)
set(UTF8PROC_INSTALL ON CACHE BOOL "" FORCE)
add_subdirectory("${SOURCES}/UTF8PROC" utf8proc)
add_library(tree-sitter STATIC "${SOURCES}/TREESITTER/lib/src/lib.c")
target_include_directories(tree-sitter PRIVATE "${SOURCES}/TREESITTER/lib/src" PUBLIC "${SOURCES}/TREESITTER/lib/include")
install(TARGETS tree-sitter ARCHIVE DESTINATION lib)
install(DIRECTORY "${SOURCES}/TREESITTER/lib/include/tree_sitter" DESTINATION include)
